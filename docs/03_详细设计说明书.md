### 2.2 任务调度器（Scheduler）

#### 2.2.1 数据结构

```rust
/// 任务元数据
struct TaskMeta {
    id: TaskId,                      // UUID v4
    name: String,
    description: Option<String>,
    source: SourceConfig,            // 数据源配置
    sink: SinkConfig,                // 目标端配置
    transform: Option<TransformConfig>,
    shard_strategy: ShardStrategy,  // 分片策略

    state: TaskState,

    updated_at: DateTime<Utc>,
    created_by: UserId,
}

/// 任务状态枚举
enum TaskState {
    Pending,
    Running { started_at: DateTime<Utc>, shard_count: u32 },
    Pausing,
    Paused,
    Canceling,
    Completed { finished_at: DateTime<Utc>, stats: TaskStats },
    Failed { error: String, failed_at: DateTime<Utc> },

/// 分片策略
enum ShardStrategy {
    ByIndexName,                    // 每个 ES 索引一个 Shard
    ByDateRange { interval: DateInterval },

/// 重试策略
struct RetryPolicy {
    max_attempts: u32,              // 最大重试次数，默认 3
    backoff: BackoffStrategy,       // 退避策略
}

    Fixed { interval_secs: u64 },
    Exponential { base_secs: u64, max_secs: u64 },
}
```

#### 2.2.2 调度流程

```rust
impl Scheduler {
    /// 提交任务（外部 API 调用入口）
    async fn submit_task(&self, task: TaskMeta) -> Result<TaskId, SchedulerError> {
        // 1. 参数校验
        self.validate_task(&task)?;
        // 2. 通过 Raft 持久化任务元数据
        self.raft.propose(RaftCommand::UpsertTask(task.clone())).await?;
        // 3. 触发分片生成
        let shards = self.shard_generator.generate(&task).await?;
        for shard in shards {
            self.raft.propose(RaftCommand::UpsertShard(shard)).await?;
        }
        // 4. 通知 Agent 有新任务可用（通过 gRPC Stream 广播）
        self.agent_notifier.broadcast_new_task(task.id).await;
        Ok(task.id)
    }

    /// 监控任务进度，处理超时和故障恢复
    async fn monitor_loop(&self) {
        loop {
            self.check_timed_out_shards().await;    // 检测超时 Shard
            self.requeue_orphan_shards().await;     // 重新入队孤儿 Shard（Agent 故障后）
            self.update_task_state().await;         // 聚合 Shard 状态更新 Task 状态
            tokio::time::sleep(Duration::from_secs(5)).await;
        }
    }
}
```

```rust
    async fn start(&self, raft_client: RaftClient, grpc_control: ControlGrpcClient) {
        for i in 0..self.config.max_concurrent {

    }
```

#### 3.1.2 Worker 任务抢占逻辑

```rust
impl Worker {
    async fn run_loop(&self) {
        loop {
            // 1. 尝试从分布式队列中抢占一个待执行 Shard
            match self.try_claim_shard().await {
                Some(shard_id) => {
                    // 2. 执行 Shard
                    let result = self.execute_shard(shard_id).await;
                    // 3. 上报结果
                    self.report_shard_result(shard_id, result).await;
                },
                None => {
                    // 没有可用 Shard，等待通知
                    self.wait_for_notification().await;
                }
            }
        }
    }

    async fn try_claim_shard(&self) -> Option<ShardId> {
        // 使用 Raft CAS（Compare-And-Swap）原子操作
        // 将 Shard 状态从 Pending → Running，并记录执行节点 ID
        self.raft.cas_shard_state(
            ShardState::Pending,
            ShardState::Running { node_id: self.node_id, started_at: Utc::now() }
        ).await.ok()
    }
}
```

### 3.2 进度追踪（Offset Tracking）

```rust
struct OffsetInfo {
    shard_id: ShardId,
    source_type: String,
    /// ES: 已处理文档数
    /// MySQL: 最大已处理主键值
    /// 文件: 已读取字节偏移
    offset: OffsetValue,
    checkpoint_at: DateTime<Utc>,
    docs_processed: u64,
    bytes_processed: u64,
}

enum OffsetValue {
    DocumentCount(u64),
    PrimaryKey(serde_json::Value),
    ByteOffset(u64),
    BinlogPosition { file: String, position: u64 },
}

impl OffsetTracker {
    /// 每处理 N 条记录或每隔 T 秒，将 Offset 持久化至 Raft
    async fn checkpoint(&self, offset: OffsetInfo) -> Result<(), Error> {
        self.raft.propose(RaftCommand::UpdateShardOffset(
            offset.shard_id,
            offset,
        )).await
    }
}
```

```rust
trait Source: Send + Sync {

    async fn connect(&self) -> Result<(), DriverError>;

    /// 根据 Offset 读取数据，返回批次
    async fn read_batch(

        offset: &OffsetValue,

    /// 获取 Shard 的预估总数据量（用于进度百分比计算）

    /// 关闭连接
    async fn close(&self) -> Result<(), DriverError>;
}

/// 数据目标 Trait
#[async_trait]
trait Sink: Send + Sync {
    async fn connect(&self) -> Result<(), DriverError>;
    async fn write_batch(&mut self, records: Vec<Record>) -> Result<WriteResult, DriverError>;
    async fn flush(&mut self) -> Result<(), DriverError>;
    async fn close(&self) -> Result<(), DriverError>;
}

/// 变换 Trait
trait Transform: Send + Sync {
    fn transform(&self, record: Record) -> Result<Vec<Record>, TransformError>;

    source_type: String,
    payload: serde_json::Value,
```

### 4.2 管道执行引擎

```rust
struct Pipeline {
    source: Box<dyn Source>,
    transforms: Vec<Box<dyn Transform>>,
    sink: Box<dyn Sink>,
    batch_size: usize,           // 每批次处理的记录数，默认 1000
    offset_checkpoint_interval: u64, // 每 N 条记录触发一次 Checkpoint
}

impl Pipeline {
    async fn run(&mut self, initial_offset: OffsetValue) -> Result<PipelineStats, Error> {
        let mut offset = initial_offset;
        let mut stats = PipelineStats::default();

        loop {
            // 1. 读取一批数据
            let (records, new_offset) = self.source.read_batch(&offset, self.batch_size).await?;
            if records.is_empty() { break; } // 数据读完

            // 2. 应用变换
            let transformed: Vec<Record> = records
                .into_iter()
                .flat_map(|r| self.transforms.iter().fold(vec![r], |acc, t| {
                    acc.into_iter().flat_map(|rec| t.transform(rec).unwrap_or_default()).collect()
                }))
                .collect();

            // 3. 写入目标端
            let write_result = self.sink.write_batch(transformed).await?;
            stats.docs_written += write_result.success_count;
            stats.bytes_written += write_result.bytes_written;

            // 4. 更新 Offset 并持久化
            offset = new_offset;
            if stats.docs_written % self.offset_checkpoint_interval == 0 {
                self.offset_tracker.checkpoint(/* offset info */).await?;
            }
        }
        self.sink.flush().await?;
        Ok(stats)
    }
}
```

```protobuf
package dataspool.v1;

service TaskService {
    // 创建任务（仅 Leader 处理）
    rpc CreateTask(CreateTaskRequest) returns (CreateTaskResponse);

    string current_offset = 5;    // JSON 序列化的 OffsetValue
    float throughput_docs_per_sec = 6;
    int64 timestamp_ms = 7;
}

message TaskCommand {
    string task_id = 1;
    oneof command {
        PauseCommand pause = 2;
        ResumeCommand resume = 3;
        CancelCommand cancel = 4;
        UpdateRateLimitCommand rate_limit = 5;
    }
}
```

```rust
struct GrpcConnectionPool {
    /// 对每个节点维护一个 gRPC Channel
    connections: HashMap<NodeId, Channel>,
    /// 连接健康检查间隔
    health_check_interval: Duration,
}

impl GrpcConnectionPool {
    /// 获取指定节点的 Channel，自动重连
    async fn get_channel(&self, node_id: &NodeId) -> Result<Channel, ConnectionError> {
        if let Some(ch) = self.connections.get(node_id) {
            if ch.is_connected() { return Ok(ch.clone()); }
        }
        self.reconnect(node_id).await
    }
}
```

```yaml
  jwt_secret: "${DATASPOOL_JWT_SECRET}"  # 支持环境变量
  token_expiry: "24h"
  api_keys: []  # 可配置静态 API Key 列表

# 可观测性
observability:
  log_level: "info"  # debug | info | warn | error
  log_format: "json"  # json | text
  metrics_enabled: true
  tracing:
    enabled: false
    endpoint: "http://jaeger:4317"

# 告警
alerting:
  rules:
    - name: "task_failed"
      condition: "task.state == Failed"
      channels: ["webhook_dingtalk"]
  channels:
    webhook_dingtalk:
      type: "webhook"
      url: "${DINGTALK_WEBHOOK_URL}"
```

### 9.2 环境变量映射

所有配置项均可通过环境变量覆盖，命名规则为：`DATASPOOL_` + 路径大写下划线连接，例如：

- `server.http_port` → `DATASPOOL_SERVER_HTTP_PORT`
- `agent.resources.max_concurrent_tasks` → `DATASPOOL_AGENT_MAX_CONCURRENT_TASKS`

## 10. 关键流程时序图

### 10.1 任务提交与分片分配流程

```
Client          API Gateway      Scheduler(Leader)    Raft Cluster     Agent(s)
  │                   │                  │                   │              │
  │─POST /tasks──────►│                  │                   │              │
  │                   │─validate()───────►                   │              │
  │                   │                  │─propose(Task)─────►              │
  │                   │                  │◄──────committed───│              │
  │                   │                  │─generate_shards()─►              │
  │                   │                  │─propose(Shards)───►              │
  │                   │                  │◄──────committed───│              │
  │                   │◄─task_id─────────│                   │              │
  │◄──task_id─────────│                  │─broadcast_new_task()────────────►│
  │                   │                  │                   │              │
  │                   │                  │             (Agent polls)        │
  │                   │                  │◄──────claimShard()───────────────│
  │                   │                  │─CAS(Pending→Running)─►           │
  │                   │                  │◄─────────────────────────shard───│
  │                   │                  │             (Agent executes)      │
  │                   │                  │◄──reportProgress(stream)─────────│
```

### 10.2 节点故障恢复流程

```
Node-1(Leader)    Node-2           Node-3            Scheduler
     │                │                │                 │
  [CRASH]             │                │                 │
     ×                │                │                 │
                      │──heartbeat────►│                 │
                      │  timeout       │                 │
                      │──elect self───►│                 │
                      │◄──vote granted─│                 │
                   [Node-2 becomes Leader]               │
                      │──requeue orphan shards──────────►│
                      │                │    (shards from Node-1 reset to Pending)
                      │──broadcast_new_task()────────────►│
                                                          │
                                                     [Node-3 claims shards]
                                                     [Continues from last offset]
```
